name: sms-spam-detector

services:
  api:
    build:
      context: .
      dockerfile: Dockerfile
    image: sms-spam-detector:latest
    container_name: sms-spam-api
    ports:
      - "8000:8000"
    environment:
      SPAM_THRESHOLD: ${SPAM_THRESHOLD:-0.5}
      SPAM_LOG_LEVEL: ${SPAM_LOG_LEVEL:-INFO}
      SPAM_MAX_BATCH_SIZE: ${SPAM_MAX_BATCH_SIZE:-500}
      # Leave empty to disable POST /admin/reload.
      SPAM_ADMIN_TOKEN: ${SPAM_ADMIN_TOKEN:-}
    volumes:
      - feedback:/app/feedback
    restart: unless-stopped
    healthcheck:
      test:
        - CMD
        - python
        - -c
        - "import urllib.request; urllib.request.urlopen('http://127.0.0.1:8000/health', timeout=3)"
      interval: 30s
      timeout: 5s
      retries: 3
      start_period: 20s

  feedback:
